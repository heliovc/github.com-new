The complete script (sort_downloads.py)
from pathlib import Path

downloads = Path("Downloads")
folders = {
    "Images": {".jpg", ".jpeg", ".png", ".gif"},
    "Documents": {".pdf", ".docx", ".txt", ".xlsx", ".csv"},
    "Audio": {".mp3", ".wav"},
    "Video": {".mp4", ".mov"},
    "Archives": {".zip", ".rar"},
}
skip = {".crdownload", ".part", ".tmp", ".ini"}

def folder_for(ext):
    for name, exts in folders.items():
        if ext in exts:
            return name
    return "Other"

def unique(dest):
    new, n = dest, 1
    while new.exists():
        new = dest.with_name(f"{dest.stem} ({n}){dest.suffix}")
        n += 1
    return new

for file in sorted(downloads.iterdir()):
    ext = file.suffix.lower()
    if not file.is_file() or file.name.startswith(".") or ext in skip:
        continue
    target = downloads / folder_for(ext)
    target.mkdir(exist_ok=True)
    dest = unique(target / file.name)
    file.rename(dest)
    print(f"{file.name} → {target.name}/{dest.name}")

Bonus: sort by month (sort_by_month.py)
from pathlib import Path
from datetime import datetime

downloads = Path("Downloads")

for file in sorted(downloads.iterdir()):
    if file.is_file() and not file.name.startswith("."):
        modified = datetime.fromtimestamp(file.stat().st_mtime)
        target = downloads / modified.strftime("%Y-%m")
        target.mkdir(exist_ok=True)
        file.rename(target / file.name)
        print(f"{file.name} → {target.name}")

Use your real folder: downloads = Path.home() / "Downloads"
Tested on Python 3.13. On Mac and Linux, rename silently replaces an existing file; on Windows it raises FileExistsError. The unique() function avoids both. Moving to a different drive? Use shutil.move instead of rename.
